        function serverBuildFighter(filler)
        {
            // Create a new temporary storage table with the required fields
              var rs = Jaxer.DB.execute("SELECT l.name FROM welcometostreetfighterlist l");          
            // Select the single record from the database
            return rs;
        }

        function serverSelectFighter(fighter)
        {
            // Create a new temporary storage table with the required fields
              var rs = Jaxer.DB.execute("SELECT l.age, l.country, l.style, l.image FROM welcometostreetfighterlist l WHERE l.name = ?",
                    [fighter]);      
            // Select the single record from the database
            var tempstorage = rs.rows[0];
            // Return the temporary storage variable to the calling function
            return tempstorage;
        }

        function serverBuildTechnique(fighter)
        {
            // Create a new temporary storage table with the required fields
            var rs = Jaxer.DB.execute("SELECT m.name FROM welcometostreetfighterlist l, " +
                        "welcometostreetfightermoveset m WHERE l.name = ? AND l.movesetid = m.movesetid", [fighter]);        
            // Select the single record from the database
            return rs;
        }

        function serverSelectTechnique(technique)
